import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:timezone/timezone.dart' as tz;
import 'package:timezone/data/latest_all.dart' as tzdata;

class NotificationService {
  static final _plugin = FlutterLocalNotificationsPlugin();

  static Future<void> initialize() async {
    tzdata.initializeTimeZones();
    tz.setLocalLocation(tz.getLocation('Asia/Bangkok'));

    const init = InitializationSettings(
      android: AndroidInitializationSettings('@mipmap/ic_launcher'),
      iOS: DarwinInitializationSettings(),
    );
    await _plugin.initialize(init, onDidReceiveNotificationResponse: _onTap);

    await _plugin
        .resolvePlatformSpecificImplementation<AndroidFlutterLocalNotificationsPlugin>()
        ?.createNotificationChannel(const AndroidNotificationChannel(
          'office_syndrome_main',
          'Office Syndrome',
          description: 'แจ้งเตือนยืดเหยียดระหว่างงาน',
          importance: Importance.high,
        ));
  }

  static Future<void> scheduleAt(DateTime when,
      {required int id,
      required String title,
      String? body,
      bool playSound = true,
      bool vibration = true}) async {
    final details = NotificationDetails(
      android: AndroidNotificationDetails(
        'office_syndrome_main',
        'Office Syndrome',
        channelDescription: 'แจ้งเตือนยืดเหยียดระหว่างงาน',
        importance: Importance.high,
        priority: Priority.high,
        playSound: playSound,
        enableVibration: vibration,
      ),
      iOS: const DarwinNotificationDetails(),
    );

    await _plugin.zonedSchedule(
      id,
      title,
      body,
      tz.TZDateTime.from(when, tz.local),
      details,
      androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
      uiLocalNotificationDateInterpretation:
          UILocalNotificationDateInterpretation.absoluteTime,
    );
  }

  static Future<void> showNow({
    required int id,
    required String title,
    String? body,
  }) async {
    await _plugin.show(
      id,
      title,
      body,
      const NotificationDetails(
        android: AndroidNotificationDetails(
          'office_syndrome_main',
          'Office Syndrome',
          importance: Importance.high,
          priority: Priority.high,
        ),
        iOS: DarwinNotificationDetails(),
      ),
    );
  }

  static Future<void> cancelAll() => _plugin.cancelAll();

  static void _onTap(NotificationResponse r) {
    // TODO: เปิดหน้า Todo/Exercise flow
  }
}

2) lib/services/background_scheduler.dart
import 'package:workmanager/workmanager.dart';
import 'package:get/get.dart';
import 'package:office_syndrome_helper/services/database_service.dart';
import 'package:office_syndrome_helper/services/notification_service.dart';
import 'package:office_syndrome_helper/controllers/notification_controller.dart';

class BackgroundScheduler {
  static bool _registered = false;

  static Future<void> ensureRegistered() async {
    if (_registered) return;
    _registered = true;

    await Workmanager().initialize(_dispatcher, isInDebugMode: false);
    await Workmanager().registerPeriodicTask(
      'osh_tick',
      'tick',
      frequency: const Duration(hours: 1),
      existingWorkPolicy: ExistingWorkPolicy.keep,
      constraints: Constraints(networkType: NetworkType.not_required),
    );
  }

  static void _dispatcher() {
    Workmanager().executeTask((task, input) async {
      final s = await DatabaseService.loadSettings();
      if (!s.notificationEnabled) return true;

      final controller = Get.isRegistered<NotificationController>()
          ? Get.find<NotificationController>()
          : NotificationController();

      final next = controller.calculateNextNotificationTime();
      await NotificationService.scheduleAt(
        next,
        id: 1001,
        title: 'ถึงเวลายืดเหยียดแล้ว',
        playSound: s.soundEnabled,
        vibration: s.vibrationEnabled,
      );
      return true;
    });
  }
}

3) lib/services/permission_service.dart
import 'package:permission_handler/permission_handler.dart';
import 'package:office_syndrome_helper/services/database_service.dart';

class PermissionService {
  static Future<bool> ensureNotificationPermissionOnce() async {
    final s = await DatabaseService.loadSettings();
    if (!s.hasRequestedPermissions) {
      final r = await Permission.notification.request(); // Android 13+
      s.hasRequestedPermissions = true;
      await DatabaseService.saveSettings(s);
      return r.isGranted || r.isLimited;
    }
    final st = await Permission.notification.status;
    return st.isGranted || st.isLimited;
  }
}
