3.1 NotificationController (คำนวณเวลาถัดไป + schedule/cancel)
// FILE: lib/controllers/notification_controller.dart
import 'dart:math';
import 'package:get/get.dart';
import '../services/database_service.dart';
import '../services/notification_service.dart';
import '../utils/date_utils.dart';
import '../models/user_settings.dart';

class NotificationController extends GetxController {
  final nextTime = Rxn<DateTime>();
  final countdownText = '—'.obs;

  /// คำนวณเวลานัดถัดไปแบบ “กระโดด boundary”
  /// - ถ้านอกเวลางาน → กระโดดไปจุดเริ่มวันทำงานถัดไป
  /// - ถ้าทับช่วงพัก → กระโดดไป endMinutes ของช่วงพักนั้น
  /// - ถ้าเกินเวลางาน → กระโดดไปวันทำงานถัดไปที่ startMinutes
  DateTime calculateNextNotificationTime({DateTime? base}) {
    final now = DateTime.now();
    final s = DatabaseService._settings!.get('user_settings') ?? UserSettings.defaults();
    final int interval = s.notificationIntervalMinutes;

    // กำหนดฐานเริ่ม: ใช้ lastNotificationTime ถ้ามี, ไม่งั้นตอนนี้
    DateTime t = (base ?? s.lastNotificationTime ?? now);

    // เริ่มจาก candidate = ล่าสุด + interval
    t = t.add(Duration(minutes: interval));

    // loop กระโดดจน valid
    for (int hop = 0; hop < 64; hop++) {
      final w = weekday1to7(t);
      if (!s.workingDays.contains(w)) {
        // ไปวันทำงานถัดไป 09:00 (workStart)
        t = _jumpToNextWorkingDayStart(t, s);
        continue;
      }
      final m = t.hour * 60 + t.minute;
      if (m < s.workStartMinutes) {
        t = DateTime(t.year, t.month, t.day)
            .add(Duration(minutes: s.workStartMinutes));
        continue;
      }
      if (m >= s.workEndMinutes) {
        t = _jumpToNextWorkingDayStart(t, s);
        continue;
      }
      // กระโดดข้ามช่วงพัก (ถ้าตกในช่วงพัก)
      final b = _findBreakIfInside(t, s);
      if (b != null) {
        final end = (b['endMinutes'] as int?) ?? s.workEndMinutes;
        t = DateTime(t.year, t.month, t.day).add(Duration(minutes: end));
        continue;
      }
      // ผ่านทุกเงื่อนไข → valid
      return t;
    }
    // fallback: ถ้ามีอะไรผิด ให้เป็น start ของวันทำงานถัดไป
    return _jumpToNextWorkingDayStart(now, s);
  }

  Future<void> rescheduleNext() async {
    final s = await DatabaseService.loadSettings();
    if (!s.notificationEnabled) {
      await NotificationService.cancelAll();
      nextTime.value = null;
      return;
    }
    final nt = calculateNextNotificationTime(
      base: s.lastNotificationTime ?? DateTime.now(),
    );
    nextTime.value = nt;
    await NotificationService.cancelAll();
    await NotificationService.scheduleAt(
      nt,
      id: 1001,
      title: 'ถึงเวลายืดเหยียดแล้ว',
      body: 'กดเพื่อเริ่มท่าออกกำลังกายสั้น ๆ',
      playSound: s.soundEnabled,
      vibration: s.vibrationEnabled,
    );
    s.lastNotificationTime = DateTime.now();
    await DatabaseService.saveSettings(s);
  }

  // --- Helpers ---
  DateTime _jumpToNextWorkingDayStart(DateTime t, UserSettings s) {
    for (int i = 1; i <= 8; i++) {
      final d = t.add(Duration(days: i));
      if (s.workingDays.contains(weekday1to7(d))) {
        return DateTime(d.year, d.month, d.day).add(
          Duration(minutes: s.workStartMinutes),
        );
      }
    }
    // ถ้าไม่เจอเลย (ไม่น่ามี) กลับวันนี้ start
    return DateTime(t.year, t.month, t.day)
        .add(Duration(minutes: s.workStartMinutes));
  }

  Map<String, dynamic>? _findBreakIfInside(DateTime t, UserSettings s) {
    final w = weekday1to7(t);
    final m = t.hour * 60 + t.minute;
    for (final bp in s.breakPeriods) {
      final days = (bp['enabledDays'] as List?)?.cast<int>() ?? <int>[];
      if (days.isNotEmpty && !days.contains(w)) continue;
      final start = bp['startMinutes'] as int? ?? -1;
      final end = bp['endMinutes'] as int? ?? -1;
      if (start >= 0 && end > start && m >= start && m < end) {
        return bp;
      }
    }
    return null;
  }

  // อัปเดตข้อความ countdown ให้หน้า Home
  void tickCountdown() {
    final nt = nextTime.value;
    if (nt == null) {
      countdownText.value = 'ปิดการแจ้งเตือน';
      return;
    }
    final diff = nt.difference(DateTime.now());
    if (diff.isNegative) {
      countdownText.value = 'กำลังถึงเวลา…';
      return;
    }
    final h = diff.inHours;
    final m = diff.inMinutes % 60;
    final s = diff.inSeconds % 60;
    countdownText.value =
        [if (h > 0) '${h}ชม', '${m}นาที', '${s}วิ'].join(' ');
  }
}

3.2 SettingsController (โหลด/บันทึกค่า + validate + trigger reschedule)
// FILE: lib/controllers/settings_controller.dart
import 'package:flutter/material.dart';
import 'package:get/get.dart';
import '../services/database_service.dart';
import '../controllers/notification_controller.dart';
import '../models/user_settings.dart';
import '../utils/date_utils.dart';
import '../utils/constants.dart';

class SettingsController extends GetxController {
  final loading = true.obs;
  late UserSettings data;

  // fields (observable)
  final notificationEnabled = true.obs;
  final soundEnabled = true.obs;
  final vibrationEnabled = true.obs;
  final interval = 60.obs;
  final workStartMinutes = (9 * 60).obs;
  final workEndMinutes = (17 * 60).obs;
  final workingDays = <int>[1,2,3,4,5].obs;
  final breakPeriods = <Map<String, dynamic>>[].obs;

  @override
  Future<void> onInit() async {
    super.onInit();
    data = await DatabaseService.loadSettings();
    notificationEnabled.value = data.notificationEnabled;
    soundEnabled.value = data.soundEnabled;
    vibrationEnabled.value = data.vibrationEnabled;
    interval.value = data.notificationIntervalMinutes;
    workStartMinutes.value = data.workStartMinutes;
    workEndMinutes.value = data.workEndMinutes;
    workingDays.assignAll(data.workingDays);
    breakPeriods.assignAll(data.breakPeriods);
    loading.value = false;
  }

  Future<void> save() async {
    // validate
    if (interval.value < 30 || interval.value > 120) {
      Get.snackbar('ข้อผิดพลาด', 'ช่วงเวลาแจ้งเตือนต้องอยู่ระหว่าง 30–120 นาที');
      return;
    }
    if (workStartMinutes.value >= workEndMinutes.value) {
      Get.snackbar('ข้อผิดพลาด', 'เวลาเริ่มต้องน้อยกว่าเวลาสิ้นสุด');
      return;
    }
    // save
    data
      ..notificationEnabled = notificationEnabled.value
      ..soundEnabled = soundEnabled.value
      ..vibrationEnabled = vibrationEnabled.value
      ..notificationIntervalMinutes = interval.value
      ..workStartMinutes = workStartMinutes.value
      ..workEndMinutes = workEndMinutes.value
      ..workingDays = workingDays.toList()
      ..breakPeriods = breakPeriods.toList();
    await DatabaseService.saveSettings(data);

    // reschedule
    final nc = Get.find<NotificationController>();
    await nc.rescheduleNext();

    Get.snackbar('บันทึกแล้ว', 'ปรับตารางแจ้งเตือนใหม่เรียบร้อย');
  }

  // helpers for UI pickers
  void pickStartTime(TimeOfDay t) => workStartMinutes.value = toMinutes(t);
  void pickEndTime(TimeOfDay t) => workEndMinutes.value = toMinutes(t);

  void toggleDay(int d) {
    final idx = workingDays.indexOf(d);
    if (idx >= 0) {
      workingDays.removeAt(idx);
    } else {
      workingDays.add(d);
      workingDays.sort();
    }
  }

  void addBreak({required int start, required int end, List<int>? days}) {
    if (end <= start) {
      Get.snackbar('ข้อผิดพลาด', 'ช่วงพักต้องสิ้นสุดหลังเวลาเริ่ม');
      return;
    }
    breakPeriods.add({
      'name': 'พัก',
      'startMinutes': start,
      'endMinutes': end,
      'enabledDays': (days ?? workingDays).toList(),
    });
  }

  void removeBreak(int index) {
    if (index >= 0 && index < breakPeriods.length) {
      breakPeriods.removeAt(index);
    }
  }
}

import 'dart:async';
import 'package:get/get.dart';
import 'notification_controller.dart';

class HomeController extends GetxController {
  Timer? _timer;
  final nc = Get.find<NotificationController>();

  @override
  void onInit() {
    super.onInit();
    nc.rescheduleNext(); // คำนวณครั้งแรก
    _timer = Timer.periodic(const Duration(seconds: 1), (_) => nc.tickCountdown());
  }

  @override
  void onClose() {
    _timer?.cancel();
    super.onClose();
  }
}
