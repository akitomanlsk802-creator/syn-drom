📋 Office Syndrome Helper - Complete Requirements Document v5.0
🎯 Executive Summary
Office Syndrome Helper เป็นแอปพลิเคชัน Android ที่ออกแบบมาเพื่อช่วยเหลือคนทำงานออฟฟิศในการป้องกันและแก้ไขปัญหา Office Syndrome ผ่านระบบแจ้งเตือนที่ฉลาดและการออกกำลังกายแบบเฉพาะเจาะจง
ปัญหาที่แก้ไข
* คนทำงานออฟฟิศมักลืมลุกขยับตัว เลยปวดเมื่อยเรื้อรัง
* ท่าออกกำลังกายที่จำเจ ทำให้เบื่อและไม่ต่อเนื่อง
* ไม่รู้ว่าปวดจุดไหนควรทำท่าไหน
* ไม่มีระบบติดตามความก้าวหน้า
โซลูชันที่นำเสนอ
1. ระบบแจ้งเตือนอัจฉริยะ - คำนวณเวลาแม่นยำ รองรับช่วงพัก
2. การสุ่มท่าออกกำลังกาย - ไม่จำเจ เฉพาะจุดที่ปวด
3. ระบบติดตามสถิติ - Charts แสดงความก้าวหน้า
4. ความยืดหยุ่น - ปรับช่วงพัก เลื่อน ข้าม ได้ตามสะดวก

UserSettings (@HiveType(typeId: 0))
เก็บค่าตั้งค่ารวมทั้งหมดของผู้ใช้ (หนึ่งแถวเดียวในกล่อง settings)
@HiveType(typeId: 0)
class UserSettings extends HiveObject {
  // จุดปวดที่เลือก (อ้างอิง id 1..8 ด้านล่าง)
  // หมายเหตุ: ฐานข้อมูล "ยืดหยุ่น" 1..8 รายการได้
  // แต่ใน Onboarding/Setup "UI จะบังคับ" ให้เลือก >= 3 เพื่อไปต่อ
  @HiveField(0)  List<int> selectedPainPointIds;


  // นาทีระหว่างการแจ้งเตือน (ช่วงที่รองรับ: 30..120)
  @HiveField(1)  int notificationIntervalMinutes;


  // เวลาทำงาน (เก็บเป็นนาทีตั้งแต่ 00:00 เพื่อง่ายต่อการคำนวณ/ค้นหา)
  @HiveField(2)  int workStartMinutes;    // default:  9*60  = 540 (09:00)
  @HiveField(3)  int workEndMinutes;      // default: 17*60  = 1020 (17:00)


  // วันทำงาน: 1=จันทร์ ... 7=อาทิตย์ (สามารถใส่ครบทั้ง 7 วัน)
  @HiveField(4)  List<int> workingDays;


  // รายการช่วงพัก — เก็บเป็น Map เพื่อหลีกเลี่ยงปัญหา codegen/adapter
  // รูปแบบแต่ละรายการ:
  // {
  //   "enabledDays": <List<int>>,      // 1..7
  //   "startMinutes": <int>,           // 0..1439
  //   "endMinutes": <int>,             // 0..1439
  //   "name": <String?>                // ชื่อช่วงพัก (optional)
  // }
  @HiveField(5)  List<Map<String, dynamic>> breakPeriods;


  // เสียง/สั่น/เปิดระบบแจ้งเตือน
  @HiveField(6)  bool soundEnabled;
  @HiveField(7)  bool vibrationEnabled;
  @HiveField(8)  bool notificationEnabled;


  // จำกัดจำนวน snooze ต่อ session (default: 3)
  @HiveField(9)  int maxSnoozeCount;


  // เคยผ่าน flow ขอสิทธิ์ (notifications ฯลฯ) แล้วหรือยัง
  @HiveField(10) bool hasRequestedPermissions;


  // เวลาที่แจ้งเตือนล่าสุด (ใช้คำนวณช่วงคงที่แบบ fixed interval)
  @HiveField(11) DateTime? lastNotificationTime;


  // อ้างอิง session ปัจจุบัน (ถ้ามี)
  @HiveField(12) String? currentSessionId;
}


ข้อกำหนด/ข้อควรระวัง
      * Validation:

         * notificationIntervalMinutes ∈ [30, 120]

         * workStartMinutes < workEndMinutes และทั้งคู่ ∈ [0, 1439]

         * workingDays ต้องมีเลขในช่วง 1..7

         * ช่วงพัก ห้ามซ้อนเวลา ภายในวันเดียวกัน (ตรวจตอนบันทึก UI/Service)

            * Mapping Utilities (แนะนำให้มีในโค้ด):

               * TimeOfDay toTimeOfDay(int minutes) / int toMinutes(TimeOfDay t)

NotificationSession (@HiveType(typeId: 1))
บันทึก “รอบแจ้งเตือน/การออกท่า” ทีละครั้ง
@HiveType(typeId: 1)
class NotificationSession extends HiveObject {
  @HiveField(0)  String id;                   // UUID
  @HiveField(1)  DateTime scheduledTime;      // เวลาที่ "ตั้ง" ไว้
  @HiveField(2)  DateTime? actualTime;        // เวลาที่ "แจ้งจริง" (หากเลื่อน/ดีเลย์)
  @HiveField(3)  int painPointId;             // จุดปวดที่ใช้ในรอบนี้ (1..8)
  @HiveField(4)  List<int> treatmentIds;      // ท่าที่สุ่มได้ (แนะนำ 2 ท่า)
  @HiveField(5)  SessionStatus status;        // สถานะรอบ
  @HiveField(6)  int snoozeCount;             // จำนวนครั้งที่ snooze จริง
  @HiveField(7)  DateTime? completedTime;     // เวลาจบการทำท่า
  @HiveField(8)  List<bool> treatmentCompleted; // เช่น [true,false] ตามจำนวนท่า
  @HiveField(9)  int durationSeconds;         // เวลาใช้งานจริงของรอบนี้ (รวมสองท่า)
}


@HiveType(typeId: 10) // แยก typeId สำหรับ enum (กันชนกับโมเดลอื่น)
enum SessionStatus {
  @HiveField(0) scheduled,
  @HiveField(1) inProgress,
  @HiveField(2) completed,
  @HiveField(3) snoozed,
  @HiveField(4) skipped,
}


ข้อกำหนด
                  * treatmentCompleted.length == treatmentIds.length

                  * เมื่อ status == completed → ต้องมี completedTime

                  * บันทึก snoozeCount ≤ UserSettings.maxSnoozeCount

DailyStats (@HiveType(typeId: 3))
สรุปสถิติรายวัน (กล่อง daily_stats)
@HiveType(typeId: 3)
class DailyStats extends HiveObject {
  @HiveField(0) DateTime date;                      // วันที่ (แนะนำ normalized: 00:00)
  @HiveField(1) int totalNotifications;             // แจ้งเตือนทั้งหมด
  @HiveField(2) int completedSessions;              // ทำเสร็จ
  @HiveField(3) int snoozedSessions;                // เลื่อน
  @HiveField(4) int skippedSessions;                // ข้าม
  @HiveField(5) Map<int, int> painPointFrequency;   // id จุดปวด → ความถี่
  @HiveField(6) int totalDurationSeconds;           // เวลารวมที่ใช้จริง


  double get successRate =>
      totalNotifications > 0 ? (completedSessions / totalNotifications) * 100 : 0.0;
}

BreakPeriod (ไม่ใช้ Hive Type / ไม่มี TypeID)
เหตุผล: ลดจุดพังจาก codegen/adapter และง่ายต่อการ migration ในอนาคต
การเก็บจริง: บันทึกเป็น Map<String, dynamic> ใน UserSettings.breakPeriods
 สัญญาโครงสร้างของ Map:
class BreakPeriod {
  final List<int> enabledDays; // 1..7
  final int startMinutes;      // 0..1439
  final int endMinutes;        // 0..1439
  final String? name;          // optional


  const BreakPeriod({
    required this.enabledDays,
    required this.startMinutes,
    required this.endMinutes,
    this.name,
  });


  // Getter อำนวยความสะดวก (UI/Service ใช้ในหน่วย TimeOfDay)
  TimeOfDay get startTime => TimeOfDay(hour: startMinutes ~/ 60, minute: startMinutes % 60);
  TimeOfDay get endTime   => TimeOfDay(hour: endMinutes ~/ 60,   minute: endMinutes % 60);


  Map<String, dynamic> toMap() => {
    'enabledDays': enabledDays,
    'startMinutes': startMinutes,
    'endMinutes': endMinutes,
    'name': name,
  };


  factory BreakPeriod.fromMap(Map<String, dynamic> m) => BreakPeriod(
    enabledDays: (m['enabledDays'] as List).cast<int>(),
    startMinutes: m['startMinutes'] as int,
    endMinutes: m['endMinutes'] as int,
    name: m['name'] as String?,
  );
}

Step 2: Permission Management (อัปเดต 2025)
ใช้ permission_handler + FLN 19.x
ไม่ต้องขอ SCHEDULE_EXACT_ALARM เพราะเราใช้ AndroidScheduleMode.exactAllowWhileIdle (ยกเว้นคุณเลือกใช้ AlarmManager เอง)
class PermissionService {
  static Future<bool> checkAndRequestPermissions() async {
    final s = await DatabaseService.loadSettings();
    if (!s.hasRequestedPermissions) {
      final ok = await _requestAllNeeded();
      await DatabaseService.saveSettings(s..hasRequestedPermissions = true);
      return ok;
    }
    return await _checkCurrentStatus();
  }


  static Future<bool> _requestAllNeeded() async {
    // Android 13+ จะถาม POST_NOTIFICATIONS แบบ runtime
    final notif = await Permission.notification.request();
    // อื่น ๆ ที่เป็น manifest permission: VIBRATE, WAKE_LOCK, RECEIVE_BOOT_COMPLETED (ไม่ต้องถาม runtime)
    return notif.isGranted || notif.isLimited;
  }


  static Future<bool> _checkCurrentStatus() async {
    final status = await Permission.notification.status;
    return status.isGranted || status.isLimited;
  }
}


// Required (ประกาศใน Manifest):
// - POST_NOTIFICATIONS (Android 13+ เป็น runtime)
// - RECEIVE_BOOT_COMPLETED (รีบูตแล้วยังทำงาน)
// - VIBRATE, WAKE_LOCK


Step 3: Pain Points Selection ⭐ (ปรับ)
                                                                  * UI ให้เลือก 1–3 จุด (ขั้นต่ำ 1, สูงสุด 3)

                                                                  * แสดง Feedback/คำแนะนำแบบเป็นมิตร

                                                                  * บันทึกลง UserSettings.selectedPainPointIds

class PainPointSelectionController extends GetxController {
  final selected = <int>{}.obs; // ชุด id: 1..8
  bool get isValid => selected.isNotEmpty && selected.length <= 3;


  void toggle(int id) {
    if (selected.contains(id)) {
      selected.remove(id);
    } else if (selected.length < 3) {
      selected.add(id);
    }
  }


  Future<void> save() async {
    if (!isValid) return;
    final s = await DatabaseService.loadSettings();
    s.selectedPainPointIds = selected.toList();
    await DatabaseService.saveSettings(s);
  }
}

2) Smart Notification System
Fixed-Interval Timing Logic (เร็วและข้าม boundary ได้)
คำนวณ “เวลาถัดไป” จาก lastNotificationTime + interval (หรือจาก now + interval ถ้ายังไม่เคยมี), แล้ว กระโดดข้าม ช่วงที่ไม่ถูกต้อง (นอกเวลางาน/ไม่ใช่วันทำงาน/ทับช่วงพัก) โดยไม่ไล่เพิ่มทีละนาที
class NotificationController extends GetxController {
  DateTime calculateNextNotificationTime() {
    final settings = Get.find<SettingsController>().settings.value;
    final interval = Duration(minutes: settings.notificationIntervalMinutes);


    final base = settings.lastNotificationTime ?? DateTime.now();
    final seed = base.add(interval);  // candidate เริ่มต้น


    return _findNextValidTime(seed, settings);
  }


  DateTime _findNextValidTime(DateTime cand, UserSettings s) {
    tz.initializeTimeZones(); // ทำครั้งเดียวใน bootstrap จริง
    final local = tz.getLocation('Asia/Bangkok');


    while (true) {
      final weekday = cand.weekday;
      if (!s.workingDays.contains(weekday)) {
        // กระโดดไปวันทำงานถัดไป 09:00 (หรือ workStartMinutes)
        cand = _jumpToNextWorkingDayStart(cand, s);
        continue;
      }


      final m = cand.hour * 60 + cand.minute;
      if (m < s.workStartMinutes) {
        cand = cand.copyWith(hour: s.workStartMinutes ~/ 60, minute: s.workStartMinutes % 60, second: 0, millisecond: 0);
      } else if (m >= s.workEndMinutes) {
        // ข้ามไปวันทำงานถัดไปตอนเริ่มงาน
        cand = _jumpToNextWorkingDayStart(cand.add(const Duration(days: 1)), s);
        continue;
      }


      // ตัดช่วงพัก (breakPeriods เก็บเป็น Map)
      final breaks = s.breakPeriods;
      final insideBreak = breaks.any((bp) {
        final days = (bp['enabledDays'] as List).cast<int>();
        final start = bp['startMinutes'] as int;
        final end   = bp['endMinutes'] as int;
        return days.contains(weekday) && _inRange(m, start, end);
      });


      if (insideBreak) {
        // กระโดดไปสิ้นสุดช่วงพักที่ชนอยู่
        final endMinute = _nearestBreakEndMinute(weekday, m, breaks);
        cand = cand.copyWith(hour: endMinute ~/ 60, minute: endMinute % 60, second: 0, millisecond: 0);
        continue;
      }


      // cand ตอนนี้ valid แล้ว
      return cand;
    }
  }


  DateTime _jumpToNextWorkingDayStart(DateTime from, UserSettings s) {
    var d = from;
    for (int i=0; i<8; i++) {
      if (s.workingDays.contains(d.weekday)) {
        return d.copyWith(
          hour: s.workStartMinutes ~/ 60,
          minute: s.workStartMinutes % 60,
          second: 0,
          millisecond: 0,
        );
      }
      d = d.add(const Duration(days: 1));
    }
    return from; // fallback
  }


  bool _inRange(int minuteOfDay, int start, int end) => start <= minuteOfDay && minuteOfDay < end;


  int _nearestBreakEndMinute(int weekday, int minuteOfDay, List<Map<String,dynamic>> breaks) {
    final ends = <int>[];
    for (final bp in breaks) {
      final days = (bp['enabledDays'] as List).cast<int>();
      if (!days.contains(weekday)) continue;
      final start = bp['startMinutes'] as int;
      final end   = bp['endMinutes'] as int;
      if (start <= minuteOfDay && minuteOfDay < end) ends.add(end);
    }
    ends.sort();
    return ends.isNotEmpty ? ends.first : minuteOfDay;
  }


  // เรียกเมื่อผู้ใช้แก้ settings → ยกเลิกทั้งหมดแล้วตั้งใหม่
  Future<void> onSettingsChanged() async {
    final s = Get.find<SettingsController>().settings.value;
    await NotificationService.cancelAll();
    final next = calculateNextNotificationTime();
    await NotificationService.scheduleAt(next, id: 1001, title: 'ถึงเวลายืดเหยียดแล้ว', body: _buildBodyFrom(s));
    await DatabaseService.saveSettings(s..lastNotificationTime = next);
  }


  String _buildBodyFrom(UserSettings s) => 'ดูแลจุดที่คุณเลือกไว้ให้ดีนะ 😊';
}

class NotificationService {

  static Future<void> initialize() async {

      android: AndroidInitializationSettings('@mipmap/ic_launcher'),
      iOS: DarwinInitializationSettings(),

    await _createChannel();
  }


  static Future<void> _createChannel() async {
    const ch = AndroidNotificationChannel(
      'office_syndrome_main',
      'Office Syndrome',
      description: 'แจ้งเตือนยืดเหยียดระหว่างงาน',
      importance: Importance.high,
    );
    await _plugin.resolvePlatformSpecificImplementation<AndroidFlutterLocalNotificationsPlugin>()
                 ?.createNotificationChannel(ch);
  }


  static Future<void> scheduleAt(DateTime when, {required int id, required String title, String? body}) async {
    await _plugin.zonedSchedule(
      id, title, body,
      tz.TZDateTime.from(when, tz.local),
      const NotificationDetails(
        android: AndroidNotificationDetails('office_syndrome_main','Office Syndrome', importance: Importance.high, priority: Priority.high),
      ),
      androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle, // ✅ ไม่ต้องขอ SCHEDULE_EXACT_ALARM
      uiLocalNotificationDateInterpretation: UILocalNotificationDateInterpretation.absoluteTime,
      matchDateTimeComponents: null,
    );
  }


  static Future<void> cancelAll() => _plugin.cancelAll();
  static void _onTap(NotificationResponse r) { /* เปิด flow ทำท่า */ }
}

class BreakTimeService {
  static Future<void> addBreakPeriod(BreakPeriod bp) async {
    final s = await DatabaseService.loadSettings();


    // 1) แปลงเป็น Map เพื่อเก็บใน Hive
    final map = bp.toMap();


    // 2) Validation: ห้ามซ้อนเวลาในวันเดียวกัน
    if (_hasConflict(map, s.breakPeriods)) {
      throw Exception('ช่วงเวลาซ้อนทับกับช่วงพักที่มีอยู่');
    }


    // 3) บันทึก + รีคอนฟิกแจ้งเตือน
    s.breakPeriods.add(map);
    await DatabaseService.saveSettings(s);
    await Get.find<NotificationController>().onSettingsChanged();
  }


  static bool _hasConflict(Map<String, dynamic> cand, List<Map<String, dynamic>> existing) {
    final candDays = (cand['enabledDays'] as List).cast<int>();
    final cs = cand['startMinutes'] as int, ce = cand['endMinutes'] as int;


    for (final e in existing) {
      final ed = (e['enabledDays'] as List).cast<int>();
      final es = e['startMinutes'] as int, ee = e['endMinutes'] as int;


      // ถ้ามีวันตรงกัน และช่วงเวลาเหลื่อมกัน ถือว่าทับ
      final dayOverlap = candDays.any(ed.contains);
      final timeOverlap = (cs < ee) && (es < ce);
      if (dayOverlap && timeOverlap) return true;
    }
    return false;
  }


  static bool isBreakTimeOptional() => true;
}

1) DatabaseService — Hive + Encryption + Minutes
                                                                                                                                          * ใช้ flutter_secure_storage เก็บคีย์เข้ารหัส Hive

                                                                                                                                          * เก็บเวลาเป็น นาทีตั้งแต่ 00:00

                                                                                                                                          * ไม่ลงทะเบียน Adapter ของ BreakPeriod (เพราะเก็บเป็น Map)

                                                                                                                                          * มี util loadOrCreateDefaults() และ saveDailyStats()

// lib/services/database_service.dart
import 'dart:math';
import 'dart:typed_data';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';
import 'package:hive_flutter/hive_flutter.dart';
import 'package:office_syndrome_helper/models/user_settings.dart';
import 'package:office_syndrome_helper/models/notification_session.dart';
import 'package:office_syndrome_helper/models/daily_stats.dart';


class DatabaseService {
  static const _settingsBox = 'settings';
  static const _sessionsBox = 'sessions';
  static const _dailyBox    = 'daily_stats';


  static Box<UserSettings>? _settings;
  static Box<NotificationSession>? _sessions;
  static Box<DailyStats>? _daily;


  static Future<void> initialize() async {
    await Hive.initFlutter();


    // Register adapters (ไม่มี BreakPeriodAdapter)
    Hive.registerAdapter(UserSettingsAdapter());
    Hive.registerAdapter(NotificationSessionAdapter());
    Hive.registerAdapter(DailyStatsAdapter());
    Hive.registerAdapter(SessionStatusAdapter()); // enum


    // Encryption key (32 bytes) from secure storage
    final storage = const FlutterSecureStorage();
    var keyBase64 = await storage.read(key: 'hive_key');
    Uint8List key;
    if (keyBase64 == null) {
      key = _randomBytes(32);
      await storage.write(key: 'hive_key', value: String.fromCharCodes(key));
    } else {
      key = Uint8List.fromList(keyBase64.codeUnits);
    }
    final cipher = HiveAesCipher(key);


    _settings = await Hive.openBox<UserSettings>(_settingsBox, encryptionCipher: cipher);
    _sessions = await Hive.openBox<NotificationSession>(_sessionsBox, encryptionCipher: cipher);
    _daily    = await Hive.openBox<DailyStats>(_dailyBox, encryptionCipher: cipher);
  }


  static Uint8List _randomBytes(int length) {
    final r = Random.secure(); return Uint8List.fromList(List<int>.generate(length, (_) => r.nextInt(256)));
  }


  // --- CRUD ---


  static Future<UserSettings> loadOrCreateDefaults() async {
    final s = _settings!.get('user_settings');
    if (s != null) return s;


    final def = UserSettings.defaults();   // ใช้ minutes + breakPeriods: List<Map>
    await _settings!.put('user_settings', def);
    return def;
  }


  static Future<UserSettings> loadSettings() async {
    return _settings!.get('user_settings') ?? UserSettings.defaults();
  }


  static Future<void> saveSettings(UserSettings settings) async {
    await _settings!.put('user_settings', settings);
  }


  static String _dateKey(DateTime d) => '${d.year}-${d.month.toString().padLeft(2,'0')}-${d.day.toString().padLeft(2,'0')}';


  static Future<DailyStats?> getDailyStats(DateTime date) async {
    return _daily!.get(_dateKey(date));
  }


  static Future<void> saveDailyStats(DailyStats s) async {
    await _daily!.put(_dateKey(s.date), s);
  }


  static Future<List<DailyStats>> getWeeklyStats([DateTime? startDate]) async {
    final start = DateTime((startDate ?? DateTime.now()).year, (startDate ?? DateTime.now()).month,
        (startDate ?? DateTime.now()).day).subtract(const Duration(days: 6));
    final stats = <DailyStats>[];
    for (var i = 0; i < 7; i++) {
      final date = start.add(Duration(days: i));
      final ds = await getDailyStats(date);
      stats.add(ds ?? DailyStats.empty(date));
    }
    return stats;
  }
}

3) BackgroundScheduler — WorkManager (แทน android_alarm_manager_plus)
                                                                                                                                                * ลงทะเบียน ครั้งเดียว ด้วย guard

                                                                                                                                                * ใช้ periodic 1 ชม. เพื่อ ensure re-schedule หลังรีบูต/เงื่อนไขอื่น

// lib/services/background_scheduler.dart
import 'package:workmanager/workmanager.dart';
import 'package:office_syndrome_helper/services/database_service.dart';
import 'package:office_syndrome_helper/services/notification_service.dart';
import 'package:office_syndrome_helper/models/user_settings.dart';
import 'package:office_syndrome_helper/controllers/notification_controller.dart';
import 'package:get/get.dart';


class BackgroundScheduler {
  static bool _registered = false;


  static Future<void> ensureRegistered() async {
    if (_registered) return;
    _registered = true;


    await Workmanager().initialize(_dispatcher, isInDebugMode: false);
    await Workmanager().registerPeriodicTask(
      'osh_tick', 'tick',
      existingWorkPolicy: ExistingWorkPolicy.keep,
      frequency: const Duration(hours: 1),
      constraints: Constraints(networkType: NetworkType.not_required),
    );
  }


  static void _dispatcher() {
    Workmanager().executeTask((task, input) async {
      final s = await DatabaseService.loadSettings();
      if (!s.notificationEnabled) return true;


      // ถ้าไม่มีนัดถัดไปในระบบ (หรือหลุด) → สร้างนัดใหม่
      final controller = Get.isRegistered<NotificationController>()
          ? Get.find<NotificationController>()
          : NotificationController();


      final next = controller.calculateNextNotificationTime();
      await NotificationService.scheduleAt(
        next,
        id: 1001,
        title: 'ถึงเวลายืดเหยียดแล้ว',
        playSound: s.soundEnabled,
        vibration: s.vibrationEnabled,
      );
      return true;
    });
  }
}

import 'package:permission_handler/permission_handler.dart';
import 'package:office_syndrome_helper/services/database_service.dart';


class PermissionService {
  static Future<bool> checkAndRequestNotificationsIfNeeded() async {
    final s = await DatabaseService.loadSettings();
    if (!s.hasRequestedPermissions) {
      final granted = await Permission.notification.request();
      s.hasRequestedPermissions = true;
      await DatabaseService.saveSettings(s);
      return granted.isGranted || granted.isLimited;
    } else {
      final st = await Permission.notification.status;
      return st.isGranted || st.isLimited;
    }
  }
}

5) BreakTimeService — BreakPeriod เป็น Map + ตรวจซ้อนเวลา
// lib/services/break_time_service.dart
import 'package:get/get.dart';
import 'package:office_syndrome_helper/services/database_service.dart';
import 'package:office_syndrome_helper/controllers/notification_controller.dart';


class BreakTimeService {
  static Future<void> addBreakPeriod({
    required List<int> enabledDays, // 1..7
    required int startMinutes,      // 0..1439
    required int endMinutes,        // 0..1439
    String? name,
  }) async {
    final s = await DatabaseService.loadSettings();
    final cand = {
      'enabledDays': enabledDays,
      'startMinutes': startMinutes,
      'endMinutes': endMinutes,
      'name': name,
    };


    if (_conflict(cand, s.breakPeriods)) {
      throw Exception('ช่วงเวลาซ้อนทับกับช่วงพักที่มีอยู่');
    }


    s.breakPeriods.add(cand);
    await DatabaseService.saveSettings(s);
    await Get.find<NotificationController>().onSettingsChanged();
  }


  static bool _conflict(Map<String, dynamic> cand, List<Map<String, dynamic>> existing) {
    final days = (cand['enabledDays'] as List).cast<int>();
    final cs = cand['startMinutes'] as int;
    final ce = cand['endMinutes'] as int;
    for (final e in existing) {
      final ed = (e['enabledDays'] as List).cast<int>();
      final es = e['startMinutes'] as int;
      final ee = e['endMinutes'] as int;
      if (days.any(ed.contains) && (cs < ee) && (es < ce)) return true;
    }
    return false;
  }
}

import 'dart:ui' as ui;
import 'dart:io';
import 'package:flutter/material.dart';
import 'package:path_provider/path_provider.dart';
import 'package:gal/gal.dart';


class ImageExportService {
  static Future<Map<String, dynamic>> generateStatsImage(Map<String, dynamic> statsData) async {
    final recorder = ui.PictureRecorder();
    final canvas = Canvas(recorder);
    const size = Size(800, 1200);


    // BG
    final bgPaint = Paint()..color = Colors.white;
    canvas.drawRect(const Rect.fromLTWH(0, 0, size.width, size.height), bgPaint);


    // Title
    final titlePainter = _text('🏆 Office Syndrome Helper 🏆',
        const TextStyle(fontSize: 36, fontWeight: FontWeight.bold, color: Colors.black87));
    titlePainter.paint(canvas, const Offset(32, 48));


    // Body
    final lines = <String>[
      'สรุปสัปดาห์นี้',
      '',
      '✅ ${statsData['completed']}/${statsData['total']} ครั้ง (${statsData['percentage']}%)',
      '🔥 ต่อเนื่อง ${statsData['streak']} วัน',
      '⏱️ ใช้เวลา ${statsData['duration']} นาที',
      '',
      '💪 จุดที่โฟกัส: ${statsData['painPoints']}',
      '',
      '#OfficeSyndromeHelper',
    ];
    double y = 120;
    for (final l in lines) {
      final p = _text(l, const TextStyle(fontSize: 24, color: Colors.black87));
      p.paint(canvas, Offset(32, y));
      y += 40;
    }


    final picture = recorder.endRecording();
    final image = await picture.toImage(size.width.toInt(), size.height.toInt());
    final byteData = await image.toByteData(format: ui.ImageByteFormat.png);
    final bytes = byteData!.buffer.asUint8List();


    // Temp preview file
    final dir = await getTemporaryDirectory();
    final name = 'osh_stats_${DateTime.now().millisecondsSinceEpoch}.png';
    final file = File('${dir.path}/$name');
    await file.writeAsBytes(bytes, flush: true);


    // Save to gallery
    final ok = await Gal.putImageBytes(bytes, album: 'Office Syndrome Helper', name: name);


    return {'previewPath': file.path, 'savedToGallery': ok == true};
  }


  static TextPainter _text(String s, TextStyle style) {
    final tp = TextPainter(text: TextSpan(text: s, style: style), textDirection: TextDirection.ltr);
    tp.layout(maxWidth: 736);
    return tp;
  }
}

7) StatisticsService — บันทึก/สรุป/เตรียมข้อมูลแชร์
// lib/services/statistics_service.dart
import 'package:office_syndrome_helper/services/database_service.dart';
import 'package:office_syndrome_helper/models/notification_session.dart';
import 'package:office_syndrome_helper/models/daily_stats.dart';
import 'package:office_syndrome_helper/services/date_utils.dart'; // สมมุติ: มี normalizeToMidnight()


class StatisticsService {
  static Future<void> recordSessionCompletion(NotificationSession session) async {
    final today = DateTime.now(); // หรือ normalize local midnight
    var daily = await DatabaseService.getDailyStats(today) ?? DailyStats.empty(today);


    daily.totalNotifications++;
    switch (session.status) {
      case SessionStatus.completed:
        daily.completedSessions++;
        daily.totalDurationSeconds += session.durationSeconds;
        break;
      case SessionStatus.snoozed:
        daily.snoozedSessions++;
        break;
      case SessionStatus.skipped:
        daily.skippedSessions++;
        break;
      default:
        break;
    }


    final cur = daily.painPointFrequency[session.painPointId] ?? 0;
    daily.painPointFrequency[session.painPointId] = cur + 1;


    await DatabaseService.saveDailyStats(daily);
  }


  static Future<Map<String, dynamic>> getWeeklySummary([DateTime? weekStart]) async {
    final stats = await DatabaseService.getWeeklyStats(weekStart);
    final totalNotifications = stats.fold(0, (a, s) => a + s.totalNotifications);
    final totalCompleted     = stats.fold(0, (a, s) => a + s.completedSessions);
    final avgSuccess = stats.isNotEmpty
        ? stats.map((s) => s.successRate).reduce((a, b) => a + b) / stats.length
        : 0.0;
    final totalDurationMin = (stats.fold(0, (a, s) => a + s.totalDurationSeconds) / 60).round();


    return {
      'totalNotifications': totalNotifications,
      'totalCompleted': totalCompleted,
      'averageSuccessRate': avgSuccess,
      'totalDurationMin': totalDurationMin,
      'dailyStats': stats,
    };
  }


  // ⭐ สร้างข้อมูลแชร์สำหรับ ImageExportService
  static Future<Map<String, dynamic>> generateShareData() async {
    final weekly = await getWeeklySummary();
    final s = await DatabaseService.loadSettings();
    final painPointNames = s.selectedPainPointIds.map((id) => PainPointsData.getName(id)).join(' | ');
    return {
      'completed': weekly['totalCompleted'],
      'total': weekly['totalNotifications'],
      'percentage': (weekly['averageSuccessRate'] as double).round(),
      'streak': await _getCurrentStreak(),            // TODO: คำนวณจาก daily_stats ต่อเนื่อง
      'duration': weekly['totalDurationMin'],
      'painPoints': painPointNames,
    };
  }


  static Future<int> _getCurrentStreak() async {
    // TODO: นับถอยหลังจากวันนี้ไปเรื่อย ๆ จนเจอวัน successRate == 0
    return 0;
  }
}

Security Measures (อัปเดตโค้ดให้ปลอดภัยขึ้น)
✅ DatabaseService — เปิด Hive แบบเข้ารหัส (Base64 key ใน Secure Storage)
// lib/services/database_service.dart
import 'dart:convert';
import 'dart:typed_data';
import 'package:flutter_secure_storage/flutter_secure_storage.dart';
import 'package:hive_flutter/hive_flutter.dart';
import 'package:office_syndrome_helper/models/user_settings.dart';
import 'package:office_syndrome_helper/models/notification_session.dart';
import 'package:office_syndrome_helper/models/daily_stats.dart';


class DatabaseService {
  static const _settingsBox = 'settings';
  static const _sessionsBox = 'sessions';
  static const _dailyBox    = 'daily_stats';


  static Future<void> initialize() async {
    await Hive.initFlutter();


    // Register adapters (เฉพาะที่ต้องใช้; BreakPeriod เก็บเป็น Map จึงไม่ต้องมี adapter)
    Hive.registerAdapter(UserSettingsAdapter());
    Hive.registerAdapter(NotificationSessionAdapter());
    Hive.registerAdapter(DailyStatsAdapter());
    Hive.registerAdapter(SessionStatusAdapter());


    // Generate or retrieve encryption key (256-bit) ที่เก็บใน Secure Storage (Base64)
    final secure = const FlutterSecureStorage();
    var b64 = await secure.read(key: 'hive_key');
    if (b64 == null) {
      final key = Hive.generateSecureKey();        // 32 bytes
      b64 = base64UrlEncode(key);
      await secure.write(key: 'hive_key', value: b64);
    }
    final keyBytes = base64Url.decode(b64);
    final cipher = HiveAesCipher(keyBytes);


    // Open encrypted boxes
    await Hive.openBox<UserSettings>(_settingsBox, encryptionCipher: cipher);
    await Hive.openBox<NotificationSession>(_sessionsBox, encryptionCipher: cipher);
    await Hive.openBox<DailyStats>(_dailyBox, encryptionCipher: cipher);


    // หมายเหตุ: ถ้ามี box อื่น (เช่น events/performance) ให้เปิดด้วย cipher เดียวกัน
  }

✅ Permission Minimization (Runtime เฉพาะที่จำเป็นจริง)
// lib/services/permission_service.dart
import 'package:permission_handler/permission_handler.dart';
import 'package:office_syndrome_helper/services/database_service.dart';


class PermissionService {
  static Future<bool> ensureNotificationPermissionOnce() async {
    final s = await DatabaseService.loadSettings();
    if (!s.hasRequestedPermissions) {
      final r = await Permission.notification.request(); // Android 13+
      s.hasRequestedPermissions = true;
      await DatabaseService.saveSettings(s);
      return r.isGranted || r.isLimited;
    }
    final st = await Permission.notification.status;
    return st.isGranted || st.isLimited;
  }
}

5) Troubleshooting Guide (technical/troubleshooting-guide.md)
ต้องมีเคสต่อไปนี้ (พร้อมวิธีแก้):
                                                                                                                                                                                                                                                         1. desugar_jdk_libs ต้อง ≥ 2.1.4
 อาการ: CheckAarMetadata แจ้งต้อง 2.1.4 แต่โปรเจ็กต์ใช้ 2.0.x
วิธีแก้: coreLibraryDesugaring "com.android.tools:desugar_jdk_libs:2.1.4" + เปิด coreLibraryDesugaringEnabled true

                                                                                                                                                                                                                                                         2. WorkManager เก่า/โค้ดฝั่ง shim ไม่รู้จัก
 อาการ: Unresolved reference 'shim' / PluginRegistrantCallback
 วิธีแก้: อัป workmanager รุ่นล่าสุดเข้ากับ Flutter embedding v2, Kotlin 1.9+, JDK 17, ลบโค้ด registrar เก่า

                                                                                                                                                                                                                                                         3. image_gallery_saver ไม่มี namespace / build fail
 วิธีแก้: เปลี่ยนเป็น gal แล้วใช้ Gal.putImageBytes(...)

                                                                                                                                                                                                                                                         4. Flutter Local Notifications พารามิเตอร์ไม่ตรง
 อาการ: uiLocalNotificationDateInterpretation ไม่พบ
วิธีแก้: อัปเป็น FLN 19.x และใช้ AndroidScheduleMode.exactAllowWhileIdle

                                                                                                                                                                                                                                                         5. fl_chart + vector_math method ไม่พบ (translateByDouble)
 วิธีแก้: อัป fl_chart เป็น 1.1.0 และให้ vector_math เวอร์ชันที่รองรับ (ล่าสุดจาก Flutter 3.32)

                                                                                                                                                                                                                                                         6. build_runner ชนกับ hive_generator
 วิธีแก้: ใช้ hive_generator ^2.x + build_runner ^2.5.x ให้ตรงกัน และ เลิกใช้ Adapter ของ BreakPeriod

                                                                                                                                                                                                                                                         7. Namespace not specified ในไลบรารีเก่า
 วิธีแก้: อัปปลั๊กอินหรือเปลี่ยนปลั๊กอิน (เช่นเปลี่ยน image lib), หรือกำหนด namespace ในโมดูลที่เกี่ยวข้อง

ทุกเคสให้แนบสคริปต์เช็กเวอร์ชัน/ตัวอย่าง build.gradle สั้น ๆ

🎯 Conclusion
Conclusion (v6 • 2025)
Office Syndrome Helper v6.0 เป็นแอปออฟไลน์ 100% ที่ช่วยคนทำงานจัดจังหวะ “พัก—ยืดเหยียด—กลับไปโฟกัส” อย่างยืดหยุ่น ใช้ง่าย ปลอดภัย และไม่รบกวนงาน โดยคงจุดเด่นของ v5 และปรับปรุงโครงสร้าง/เทคนิคให้ทันสมัยขึ้น
Key Success Factors
                                                                                                                                                                                                                                                                                                                     * ความยืดหยุ่น — เลือก 1–3 จุดปวด, เลือกวันทำงานได้ ครบ 7 วัน, ช่วงพักไม่บังคับ

                                                                                                                                                                                                                                                                                                                     * ความเป็นมิตร — ข้อความ “พร้อมดูแลคุณ”, โทน UX อบอุ่น, ยืนยันการเปลี่ยนค่าอย่างเข้าใจง่าย

                                                                                                                                                                                                                                                                                                                     * ความเรียบง่าย — แชร์ แบบรูปภาพอย่างเดียว (ไม่มีโซเชียลอินทิเกรต)

                                                                                                                                                                                                                                                                                                                     * Fixed Interval Logic — เวลาแจ้งเตือนแม่นยำ ±30 วินาที ภายใต้ Doze/idle

                                                                                                                                                                                                                                                                                                                     * Smart Randomization — สุ่มท่า 2 ท่า ไม่ซ้ำกันในรอบเดียว และสัมพันธ์กับจุดที่เลือก

                                                                                                                                                                                                                                                                                                                     * Data-Driven Insights — สถิติรายวัน/สัปดาห์, อัตราสำเร็จ, ความถี่จุดปวด, สตรีค

1) Data & Migration (สำคัญมาก)
                                                                                                                                                                                                                                                                                                                                                                                    * Migration v5 → v6: เพราะเราเปลี่ยนโครงข้อมูล (เช่น TimeOfDay → นาที int, BreakPeriod → Map) ห้ามแก้ typeId เดิมทับ ให้:

                                                                                                                                                                                                                                                                                                                                                                                       * เปิด box ใหม่ ชั่วคราว → อ่านค่าจากของเดิม → แปลง → เขียนลง box ใหม่ → ลบของเดิม/รีเนม

                                                                                                                                                                                                                                                                                                                                                                                       * เก็บ schemaVersion ใน box meta แล้วรัน migration หนึ่งครั้งใน DatabaseService.initialize()

                                                                                                                                                                                                                                                                                                                                                                                          * เพิ่ม MigrationPlan.md อธิบาย step-by-step และ rollback
